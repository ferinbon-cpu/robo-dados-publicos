name: TASK 112 official PPA 2018-2021 OCR live once
run-name: "TASK 112 one bounded OCR of official PPA 2018-2021 by @${{ github.actor }}"

on:
  push:
    branches:
      - task-112-real-ppa-2018-2021-ocr
    paths:
      - '.github/workflows/task-112-real-ppa-ocr-once.yml'

permissions:
  contents: read

concurrency:
  group: task-112-real-ppa-ocr-once
  cancel-in-progress: false

jobs:
  bounded-real-source-ocr:
    runs-on: ubuntu-24.04
    timeout-minutes: 25
    steps:
      - name: Checkout exact branch head
        uses: actions/checkout@de0fac2e4500dabe0009e67214ff5f5447ce83dd
        with:
          persist-credentials: false

      - name: Python 3.12
        uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97
        with:
          python-version: '3.12'
          check-latest: false

      - name: Install exact proven OCR dependencies
        shell: bash
        run: |
          set -euo pipefail
          sudo apt-get update
          sudo apt-get install -y --no-install-recommends \
            poppler-utils=24.02.0-1ubuntu9.9 \
            tesseract-ocr=5.3.4-1build5 \
            tesseract-ocr-por=1:4.1.0-2
          python -m pip install --disable-pip-version-check --no-deps 'pypdf==6.10.0'

      - name: Verify exact OCR dependency versions
        shell: bash
        run: |
          set -euo pipefail
          test "$(dpkg-query -W -f='${Version}' poppler-utils)" = "24.02.0-1ubuntu9.9"
          test "$(dpkg-query -W -f='${Version}' tesseract-ocr)" = "5.3.4-1build5"
          test "$(dpkg-query -W -f='${Version}' tesseract-ocr-por)" = "1:4.1.0-2"
          tesseract --list-langs | grep -Fx 'por'
          python - <<'PY'
          from pathlib import Path
          import json
          from robo_dados_publicos.research.ppa_2018_ocr import load_contract
          contract = load_contract(Path("config/task112_real_ppa_2018_2021_ocr.v1.json"))
          print(json.dumps({
              "schema": contract["schema"],
              "url": contract["source"]["url"],
              "max_requests": contract["source"]["max_http_requests_total"],
              "max_pages": contract["document"]["max_pdf_pages"],
              "retry": contract["source"]["retry"],
              "discovery_search": contract["source"]["discovery_search"],
          }, sort_keys=True))
          PY

      - name: Execute exactly one bounded real-source OCR
        shell: bash
        run: |
          set -euo pipefail
          rm -rf runtime/task112 runtime/task112_result.json
          python scripts/run_task112_real_ppa_ocr_once.py

      - name: Emit sanitized TASK 112 result
        if: always()
        shell: bash
        run: |
          set -euo pipefail
          echo TASK112_RESULT_JSON_BEGIN
          cat runtime/task112_result.json
          echo TASK112_RESULT_JSON_END

      - name: Upload sanitized TASK 112 evidence
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: task-112-real-ppa-ocr-result
          path: runtime/task112_result.json
          if-no-files-found: error
          retention-days: 1
