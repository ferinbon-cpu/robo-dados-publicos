          import hashlib
          import json
          from pathlib import Path
          from urllib.parse import urlparse

          from robo_dados_publicos.reconciliation.resolvers import LimeiraContractsResolver

          ALLOWED_HOST = "serv42.limeira.sp.gov.br"
          MAX_REQUESTS = 3
          task = {
              "task_id": "RECTASK_bf460d6fbffa22124902553f",
              "target_source": "LIMEIRA_CONTRATOS",
              "task_type": "FIND_CONTRACT_RECORD",
              "status": "READY_SEARCH",
              "match_keys": {
                  "year": 2025,
                  "contract_number": "10/2025.",
                  "cnpj": "59448656000114",
              },
              "search_hints": {
                  "process_number": "32.754/2025.",
              },
          }

          class BoundedResolver(LimeiraContractsResolver):
              def __init__(self):
                  super().__init__()
                  self.request_log = []

              def _request(self, url, *, method="GET", params=None):
                  if len(self.request_log) >= MAX_REQUESTS:
                      raise RuntimeError("STOP_TASK089_HTTP_BUDGET_EXCEEDED")
                  parsed = urlparse(url)
                  host = (parsed.hostname or "").lower()
                  if host != ALLOWED_HOST:
                      raise RuntimeError("STOP_TASK089_ORIGIN_OUTSIDE_ALLOWLIST")
                  self.request_log.append({
                      "ordinal": len(self.request_log) + 1,
                      "method": method.upper(),
                      "host": host,
                      "path": parsed.path,
                      "submitted_field_names": sorted((params or {}).keys()),
                  })
                  return super()._request(url, method=method, params=params)

          resolver = BoundedResolver()
          out = {
              "task": "TASK_089_BOUNDED_NEXT_EXECUTABLE_CONTRACT_QUERY",
              "base_main_sha": "a9c56592dabb5900b6389cfa138ef1d3a71b167e",
              "pre_run_task_contract_sha": "a1efbc2960012cc7fcda456beaf2f100e4fcfdbc",
              "selected_task_id": task["task_id"],
              "max_http_requests": MAX_REQUESTS,
              "hard_boundaries": {
                  "drive_reads": 0,
                  "drive_writes": 0,
                  "state_registry_writes": 0,
                  "queue_writes": 0,
                  "serving_writes": 0,
                  "publications": 0,
                  "contract_identity_assertions": 0,
                  "financial_identity_assertions": 0,
              },
          }
          try:
              result = resolver.resolve(task)
              payload = result.to_dict()
              allowed = {
                  "MATCH_CANDIDATE",
                  "NO_MATCH",
                  "STOP_CONTRACT_FORM_UNPROVEN",
                  "STOP_MISSING_CONTRACT_OR_SUPPLIER_KEY",
                  "STOP_CONTRACT_RELAY_ORIGIN_UNPROVEN",
                  "STOP_CONTRACT_RESULT_SCHEMA_UNPROVEN",
              }
              if payload["status"] not in allowed:
                  raise RuntimeError("STOP_TASK089_UNEXPECTED_RESOLVER_STATUS")
              out["resolver_result"] = payload
              out["result"] = "PASS_TASK089_BOUNDED_REMOTE_EXECUTION_RECORDED"
          except Exception as exc:
              out["resolver_result"] = {
                  "status": "STOP_TASK089_REMOTE_EXECUTION",
                  "error_type": type(exc).__name__,
                  "error": str(exc)[:500],
                  "candidates": [],
              }
              out["result"] = "STOP_TASK089_REMOTE_EXECUTION_RECORDED"

          out["request_count"] = len(resolver.request_log)
          out["requests"] = resolver.request_log
          if out["request_count"] > MAX_REQUESTS:
              raise SystemExit("STOP_TASK089_HTTP_BUDGET_EXCEEDED")

          candidate_summaries = []
          for candidate in out["resolver_result"].get("candidates") or []:
              canonical = json.dumps(candidate, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
              candidate_summaries.append({
                  "sha256": hashlib.sha256(canonical.encode("utf-8")).hexdigest(),
                  "match_signals": candidate.get("match_signals") or [],
              })
          out["candidate_summaries"] = candidate_summaries

          target = Path("runtime/task089_result.json")
          target.write_text(json.dumps(out, ensure_ascii=False, indent=2, sort_keys=True) + "\n", encoding="utf-8")
          canonical = json.dumps(out, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n"
          print(json.dumps({
              "result": out["result"],
              "resolver_status": out["resolver_result"]["status"],
              "candidate_count": len(out["resolver_result"].get("candidates") or []),
              "candidate_summaries": candidate_summaries,
              "request_count": out["request_count"],
              "canonical_sha256": hashlib.sha256(canonical.encode("utf-8")).hexdigest(),
          }, ensure_ascii=False, sort_keys=True))
          PY
