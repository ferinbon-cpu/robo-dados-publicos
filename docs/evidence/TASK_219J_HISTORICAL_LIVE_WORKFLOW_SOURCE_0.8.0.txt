          def classify_value(value):
              s = value or ""
              if s == "":
                  return "EMPTY"
              if re.fullmatch(r"(?i:true|false|yes|no|sim|nao|não)", s):
                  return "BOOLEAN_WORD"
              if re.fullmatch(r"-?\d+", s):
                  return "INTEGER"
              if re.fullmatch(r"-?\d+[.,]\d+", s):
                  return "DECIMAL"
              if re.fullmatch(r"(?i)[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}", s):
                  return "UUID_LIKE"
              if re.fullmatch(r"[A-Za-z0-9_.:/-]{1,40}", s):
                  return "SHORT_TOKEN"
              if all(ch.isprintable() for ch in s):
                  return "TEXT"
              return "OTHER"

          def sensitive_name(name):
              low = (name or "").casefold()
              return any(x in low for x in SENSITIVE)

          def field_profile(name, value):
              item = {
                  "value_class": classify_value(value),
                  "sensitive_name": sensitive_name(name),
              }
              if not item["sensitive_name"]:
                  item["value_length"] = len(value or "")
              return item

          def parse_post(raw, content_type):
              raw = raw or ""
              ctype = (content_type or "").lower()
              out = {"encoding_family": "UNKNOWN", "fields": []}
              try:
                  if "json" in ctype or raw.lstrip().startswith(("{","[")):
                      obj = json.loads(raw)
                      out["encoding_family"] = "JSON"
                      if isinstance(obj, dict):
                          for k, v in obj.items():
                              sv = "" if v is None else str(v)
                              out["fields"].append({"name": str(k), **field_profile(str(k), sv)})
                      else:
                          out["json_top_type"] = type(obj).__name__
                      return out
              except Exception:
                  pass
              try:
                  pairs = parse_qsl(raw, keep_blank_values=True, strict_parsing=False)
              except Exception:
                  pairs = []
              if pairs:
                  out["encoding_family"] = "FORM_URLENCODED_OR_QUERYLIKE"
                  for k, v in pairs:
                      out["fields"].append({"name": k, **field_profile(k, v)})
                  return out
              if raw:
                  out["encoding_family"] = "OPAQUE_NONEMPTY"
                  out["opaque_length"] = len(raw)
              else:
                  out["encoding_family"] = "EMPTY"
              return out

          def safe_structural_identifier(value):
              s = value or ""
              if not re.fullmatch(r"[A-Za-z_][A-Za-z0-9_.:-]{0,79}", s):
                  return None
              if re.search(r"\d{7,}", s):
                  return None
              if any(x in s.casefold() for x in SENSITIVE):
                  return None
              return s

          class StructureParser(HTMLParser):
              def __init__(self):
                  super().__init__(convert_charrefs=True)
                  self.tags = Counter()
                  self.attrs = Counter()
                  self.controls = set()
                  self.safe_ids = set()
              def handle_starttag(self, tag, attrs):
                  t = tag.lower()
                  self.tags[t] += 1
                  amap = {}
                  for k, v in attrs:
                      kk = (k or "").lower()
                      self.attrs[kk] += 1
                      amap[kk] = v
                  if t in {"input","select","textarea","button","form"}:
                      name = amap.get("name")
                      typ = amap.get("type")
                      safe_name = safe_structural_identifier(name)
                      safe_type = safe_structural_identifier(typ)
                      if safe_name or safe_type:
                          self.controls.add((t, safe_name, safe_type))
                  sid = safe_structural_identifier(amap.get("id"))
                  if sid:
                      self.safe_ids.add(sid)

          def sanitize_response_body(body_text, base64_encoded, mime):
              if base64_encoded:
                  raw = base64.b64decode(body_text.encode("ascii"), validate=False)
                  text = raw.decode("utf-8", errors="replace")
              else:
                  text = body_text
                  raw = text.encode("utf-8")
              out = {
                  "body_bytes": len(raw),
                  "body_sha256": hashlib.sha256(raw).hexdigest(),
                  "mime_type": mime,
              }
              if "html" in (mime or "").lower():
                  parser = StructureParser()
                  parser.feed(text)
                  out.update({
                      "tag_counts": dict(sorted(parser.tags.items())),
                      "attribute_name_counts": dict(sorted(parser.attrs.items())),
                      "controls": [
                          {"tag": t, "name": n, "type": ty}
                          for t, n, ty in sorted(parser.controls, key=lambda x: (x[0], x[1] or "", x[2] or ""))
                      ][:150],
                      "safe_structural_ids": sorted(parser.safe_ids)[:150],
                      "safe_structural_ids_truncated": len(parser.safe_ids) > 150,
                  })
              return out
