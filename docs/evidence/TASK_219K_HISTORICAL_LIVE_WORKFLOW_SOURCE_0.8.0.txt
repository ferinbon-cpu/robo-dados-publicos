          def try_base64(text):

          def encoding_class(raw):

                  return "UTF8_BOM"

                  return "UTF16_BOM"

              try:
                  raw.decode("utf-8")
                  return "UTF8"
              except Exception:
                  pass
              try:

                  return "UTF16_COMPATIBLE"
              except Exception:

                  return "LATIN1_TEXTLIKE" if nul < 0.05 else "BINARY_OR_MULTIBYTE"
