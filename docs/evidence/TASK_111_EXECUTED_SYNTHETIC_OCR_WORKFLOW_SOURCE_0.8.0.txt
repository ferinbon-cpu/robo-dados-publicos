jobs:

    timeout-minutes: 10
    steps:
      - name: Checkout exact branch head

      - name: Python 3.12

      - name: Install exact allowed OCR route dependencies

      - name: Verify exact installed package versions

          test "$(dpkg-query -W -f='${Version}' tesseract-ocr)" = "5.3.4-1build5"

          tesseract --list-langs | grep -Fx 'por'

      - name: Build image-only PDF and prove OCR route
        shell: bash
        run: |
          set -euo pipefail
          python - <<'PY'
          from __future__ import annotations
          import base64
          import hashlib
          import json
          import subprocess
          import tempfile
          import unicodedata
          from pathlib import Path

          from pypdf import PdfReader

          marker = "TASK111 MARCADOR OCR 73159"
          phrase = "EDUCAÇÃO INTEGRAL LIMEIRA"

          def sha256_path(path: Path) -> str:
              return hashlib.sha256(path.read_bytes()).hexdigest()

          def norm(text: str) -> str:
              text = unicodedata.normalize("NFKD", text)
              text = "".join(ch for ch in text if not unicodedata.combining(ch))
              return " ".join(text.upper().split())

          with tempfile.TemporaryDirectory() as td:
              root = Path(td)
              html = root / "source.html"
              html.write_text(
                  "<!doctype html><html><body style='margin:0;background:white;'>"
                  "<div style='font-family:Arial,sans-serif;font-size:52px;font-weight:700;"
                  "color:black;padding:100px 60px;line-height:1.5;'>"
                  f"<div>{marker}</div><div>{phrase}</div>"
                  "</div></body></html>",
                  encoding="utf-8",
              )

              png = root / "source.png"
              subprocess.run([
                  "google-chrome","--headless","--no-sandbox","--disable-gpu",
                  "--window-size=1600,900",
                  f"--screenshot={png}",
                  html.resolve().as_uri(),
              ], check=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)

              image_html = root / "image.html"
              image_html.write_text(
                  "<!doctype html><html><body style='margin:0;background:white;'>"
                  f"<img src='{png.resolve().as_uri()}' style='width:100%;height:auto;'>"
                  "</body></html>",
                  encoding="utf-8",
              )

              pdf = root / "image-only.pdf"
              subprocess.run([
                  "google-chrome","--headless","--no-sandbox","--disable-gpu",
                  f"--print-to-pdf={pdf}",
                  "--no-pdf-header-footer",
                  image_html.resolve().as_uri(),
              ], check=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)

              reader = PdfReader(str(pdf))
              extracted = "\n".join((p.extract_text() or "") for p in reader.pages)
              if extracted.strip():
                  print("TASK111_UNEXPECTED_PDF_TEXT=" + repr(extracted))
                  raise SystemExit("TASK111_SYNTHETIC_PDF_NOT_IMAGE_ONLY")
              if len(reader.pages) != 1:
                  raise SystemExit("TASK111_SYNTHETIC_PDF_PAGE_COUNT")

              out_prefix = root / "rendered-page-1"
              subprocess.run([
                  "pdftoppm","-f","1","-l","1","-singlefile","-r","300","-gray","-png",
                  str(pdf),str(out_prefix)
              ], check=True)
              rendered = root / "rendered-page-1.png"
              if not rendered.exists():
                  raise SystemExit("TASK111_RENDER_MISSING")

              proc = subprocess.run([
                  "tesseract",str(rendered),"stdout","-l","por","--oem","1","--psm","3","tsv"
              ], check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
              tsv = proc.stdout
              tsv_path = root / "ocr.tsv"
              tsv_path.write_text(tsv, encoding="utf-8")

              rows = [line.split("\t") for line in tsv.splitlines()[1:] if line.strip()]
              words = []
              confs = []
              for cols in rows:
                  if len(cols) < 12:
                      continue
                  text = cols[11].strip()
                  if text:
                      words.append(text)
                      try:
                          confs.append(float(cols[10]))
                      except ValueError:
                          pass

              reconstructed = norm(" ".join(words))
              if norm(marker) not in reconstructed:
                  raise SystemExit("TASK111_MARKER_NOT_RECOVERED")
              if norm(phrase) not in reconstructed:
                  raise SystemExit("TASK111_PORTUGUESE_PHRASE_NOT_RECOVERED")
              usable_confs = [c for c in confs if c >= 0]
              if not usable_confs:
                  raise SystemExit("TASK111_NO_WORD_CONFIDENCE")

              result = {
                  "schema":"TASK111_SYNTHETIC_OCR_ROUTE_PROOF_V1",
                  "status":"PASS",
                  "installed_packages":{
                      "poppler-utils":"24.02.0-1ubuntu9.9",
                      "tesseract-ocr":"5.3.4-1build5",
                      "tesseract-ocr-por":"1:4.1.0-2",
                      "pypdf":"6.10.0",
                  },
                  "pdf_page_count":1,
                  "pdf_text_empty":True,
                  "normalized_ocr_text":reconstructed,
                  "marker_recovered":True,
                  "portuguese_phrase_recovered":True,
                  "word_confidence_count":len(usable_confs),
                  "word_confidence_min":min(usable_confs),
                  "word_confidence_max":max(usable_confs),
                  "sha256":{
                      "synthetic_pdf":sha256_path(pdf),
                      "rendered_page_1":sha256_path(rendered),
                      "ocr_tsv":sha256_path(tsv_path),
                  },
                  "official_source_http_requests":0,
                  "real_ppa_reads":0,
                  "real_ppa_ocr":False,
                  "drive_reads":0,
                  "drive_writes":0,
                  "publication":False,
              }
              print("TASK111_SYNTHETIC_OCR_JSON_BEGIN")
              print(json.dumps(result, ensure_ascii=False, sort_keys=True))
              print("TASK111_SYNTHETIC_OCR_JSON_END")
          PY
