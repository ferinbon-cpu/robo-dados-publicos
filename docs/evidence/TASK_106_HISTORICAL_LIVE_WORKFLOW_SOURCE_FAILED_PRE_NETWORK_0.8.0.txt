name: TASK 106 historical PPA primary evidence live once
run-name: "TASK 106 one bounded historical PPA acquisition with pypdf by @${{ github.actor }}"

on:
  push:
    branches:
      - task-106-historical-ppa-live-pypdf
    paths:
      - '.github/workflows/task-106-historical-ppa-live-once.yml'

permissions:
  contents: read

concurrency:
  group: task-106-historical-ppa-live-once
  cancel-in-progress: false

jobs:
  bounded-readonly-primary-ppa-acquisition:
    runs-on: ubuntu-latest
    timeout-minutes: 8
    steps:
      - name: Checkout exact branch head
        uses: actions/checkout@de0fac2e4500dabe0009e67214ff5f5447ce83dd
        with:
          persist-credentials: false

      - name: Python 3.12
        uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97
        with:
          python-version: '3.12'
          check-latest: false

      - name: Install exact pinned pypdf runtime dependency
        shell: bash
        run: |
          set -euo pipefail
          python -m pip install --disable-pip-version-check --no-deps 'pypdf==6.10.0'

      - name: Prove local pypdf text extraction before any source request
        shell: bash
        run: |
          set -euo pipefail
          python - <<'PY'
          from pathlib import Path
          import tempfile
          from robo_dados_publicos.research.local_pdf_capability import (
              _minimal_pdf_bytes,
              extract_pdf_text_pypdf,
          )
          marker = "TASK106_LIVE_PREFLIGHT_PYPDF_MARKER_64271"
          with tempfile.TemporaryDirectory() as td:
              path = Path(td) / "synthetic.pdf"
              path.write_bytes(_minimal_pdf_bytes(marker))
              text = extract_pdf_text_pypdf(path)
              if marker not in text:
                  raise SystemExit("TASK106_PYPDF_PREFLIGHT_MARKER_NOT_RECOVERED")
          print("TASK106_PYPDF_PREFLIGHT_PASS")
          PY

      - name: Execute exactly one bounded read-only historical PPA acquisition
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p runtime
          python scripts/run_task106_historical_ppa_live_once.py

      - name: Emit sanitized result for immutable run-log readback
        if: always()
        shell: bash
        run: |
          set -euo pipefail
          echo TASK106_RESULT_JSON_BEGIN
          cat runtime/task106_result.json
          echo TASK106_RESULT_JSON_END

      - name: Upload sanitized bounded evidence
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: task-106-historical-ppa-primary-evidence
          path: runtime/task106_result.json
          if-no-files-found: error
          retention-days: 1
