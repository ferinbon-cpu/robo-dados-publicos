              s=unicodedata.normalize("NFKD",str(s or ""))

              s=s.casefold()
              return re.sub(r"[^a-z0-9]+"," ",s).strip()

          def scalar_type(v):
              if v is None: return "NULL"
              if isinstance(v,bool): return "BOOL"
              if isinstance(v,int) and not isinstance(v,bool): return "INT"
              if isinstance(v,float): return "DECIMAL"
              if isinstance(v,str): return "TEXT"

              raw=json.dumps(v,ensure_ascii=False,sort_keys=True,separators=(",",":"),default=str).encode("utf-8")
              return hashlib.sha256(raw).hexdigest()

          def decode_maybe_json(v, depth=0):
              if depth>3: return v

                  s=v.strip()

                      try:
                          return decode_maybe_json(json.loads(s),depth+1)
                      except Exception:
                          return v

          def canonical_key(k):
              s=str(k)

              if low.startswith("gxtpr_"):

              return re.sub(r"[^a-z0-9]","",s.casefold())

          def get_field(obj,name,default=None):

              return default

          def public_scalar(v):

              if len(v)>200 or re.fullmatch(r"[A-Za-z0-9_\-+/=]{48,}",s or ""):
                  return {"class":"TEXT_HASHED","length":len(v),"sha256":hashlib.sha256(v.encode("utf-8",errors="replace")).hexdigest()}
              return {"class":"TEXT","value":v,"length":len(v)}

          def schema_only(v,depth=0):
              v=decode_maybe_json(v)
              if depth>=8:
                  return {"type":"DEPTH_LIMIT"}
              if isinstance(v,dict):
                  fields={}
                  for k in sorted(v,key=lambda x:str(x))[:300]:
                      ks=str(k)
                      if re.fullmatch(r"[A-Za-z_][A-Za-z0-9_.:\-]{0,119}",ks):
                          key=ks
                      else:
                          key="KEY_SHA256_"+hashlib.sha256(ks.encode("utf-8",errors="replace")).hexdigest()[:16]
                      fields[key]=schema_only(v[k],depth+1)
                  return {"type":"OBJECT","key_count":len(v),"fields":fields,"truncated":len(v)>300}
              if isinstance(v,list):
                  samples=[schema_only(x,depth+1) for x in v[:100]]
                  uniq=sorted(set(json.dumps(x,ensure_ascii=False,sort_keys=True,separators=(",",":")) for x in samples))
                  return {"type":"LIST","length":len(v),"unique_element_schema_count":len(uniq),"element_schemas":[json.loads(x) for x in uniq[:20]],"truncated":len(v)>100}
              out={"type":scalar_type(v)}
              if isinstance(v,str):
                  out["length"]=len(v)
              return out

          def matched_terms(value):
              if not isinstance(value,str):

              return sorted({t for t in STRONG+SUPPORT if norm(t) in n})

          def scan_structured_terms(v,path="$",depth=0,out=None):

              elif isinstance(v,list):
                  for i,val in enumerate(v[:100]):

              elif isinstance(v,str):
                  mt=matched_terms(v)
                  if mt:
                      out.append({"path":path,"terms":mt})
              return out

          def sanitize_area(area,layer_index,area_index):
