name: TASK 114 ontology-aware PPA 2018-2021 OCR live once
run-name: "TASK 114 one ontology-aware OCR candidate scan by @${{ github.actor }}"

on:
  push:
    branches:
      - task-114-ontology-aware-ppa-ocr-live
    paths:
      - '.github/workflows/task-114-ontology-ppa-ocr-once.yml'

permissions:
  contents: read

concurrency:
  group: task-114-ontology-ppa-ocr-once
  cancel-in-progress: false

jobs:
  bounded-ontology-ocr:
    runs-on: ubuntu-24.04
    timeout-minutes: 25
    steps:
      - name: Checkout exact branch head
        uses: actions/checkout@de0fac2e4500dabe0009e67214ff5f5447ce83dd
        with:
          persist-credentials: false

      - name: Python 3.12
        uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97
        with:
          python-version: '3.12'
          check-latest: false

      - name: Install exact proven OCR dependencies
        shell: bash
        run: |
          set -euo pipefail
          sudo apt-get update
          sudo apt-get install -y --no-install-recommends \
            poppler-utils=24.02.0-1ubuntu9.9 \
            tesseract-ocr=5.3.4-1build5 \
            tesseract-ocr-por=1:4.1.0-2
          python -m pip install --disable-pip-version-check --no-deps 'pypdf==6.10.0'

      - name: Verify exact ontology and OCR preconditions
        shell: bash
        run: |
          set -euo pipefail
          test "$(dpkg-query -W -f='${Version}' poppler-utils)" = "24.02.0-1ubuntu9.9"
          test "$(dpkg-query -W -f='${Version}' tesseract-ocr)" = "5.3.4-1build5"
          test "$(dpkg-query -W -f='${Version}' tesseract-ocr-por)" = "1:4.1.0-2"
          tesseract --list-langs | grep -Fx 'por'
          python - <<'PY'
          import json
          from pathlib import Path
          from robo_dados_publicos.research.ppa_2018_ontology_search import load_and_validate_task113_contract
          result = load_and_validate_task113_contract(Path("config/task113_ppa2018_ontology_lexical.v1.json"))
          if result["term_count"] != 29 or result["family_count"] != 3:
              raise SystemExit("TASK114_ONTOLOGY_CONTRACT_DRIFT")
          print(json.dumps(result, sort_keys=True))
          PY

      - name: Execute exactly one ontology-aware OCR candidate scan
        shell: bash
        run: |
          set -euo pipefail
          rm -rf runtime/task114 runtime/task114_result.json
          python scripts/run_task114_ontology_ppa_ocr_once.py

      - name: Emit TASK 114 summary and bounded candidate list
        if: always()
        shell: bash
        run: |
          set -euo pipefail
          echo TASK114_RESULT_JSON_BEGIN
          cat runtime/task114_result.json
          echo TASK114_RESULT_JSON_END

      - name: Upload sanitized TASK 114 result
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: task-114-ontology-ppa-ocr-result
          path: runtime/task114_result.json
          if-no-files-found: error
          retention-days: 1
