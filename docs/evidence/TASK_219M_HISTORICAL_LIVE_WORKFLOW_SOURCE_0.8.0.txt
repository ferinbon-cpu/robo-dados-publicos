          import hashlib
          import json
          import os
          import re

          HOST = "transparencia.limeira.sp.gov.br"
          PATHS = [
              "/aportalclientjs.aspx",
              "/tdaportalclient.js",
              "/mp_tda.js",
              "/gxcfg.js",
              "/gxgral.js",
              "/gxtimezone.js",
          ]
          TARGETS = [
              "LayerInfo",
              "awsgetcontentareas.aspx",

              "changetitle",

          ]

          class NoRedirect(urllib.request.HTTPRedirectHandler):
              def redirect_request(self, req, fp, code, msg, headers, newurl):
                  return None

          opener = urllib.request.build_opener(NoRedirect)
          request_count = 0

          def safe_name(path):
              return path.strip("/").replace("/", "_") or "root"

          def line_of(text, pos):
              return text.count("\n", 0, pos) + 1

          def bounded_context(text, start, end, radius=320):
              lo=max(0,start-radius); hi=min(len(text),end+radius)
              s=text[lo:hi]
              s=re.sub(r"[\r\n\t]+"," ",s)
              s=re.sub(r"\s+"," ",s)
              return s[:800]

          def symbol_window(text, pos):
              lo=max(0,pos-1200); hi=min(len(text),pos+1200)
              chunk=text[lo:hi]
              fn_patterns=[
                  r"function\s+([A-Za-z_$][A-Za-z0-9_$]*)\s*\(",
                  r"([A-Za-z_$][A-Za-z0-9_$.]*)\s*=\s*function\s*\(",
                  r"([A-Za-z_$][A-Za-z0-9_$.]*)\s*:\s*function\s*\("
              ]
              names=[]
              for pat in fn_patterns:
                  for m in re.finditer(pat,chunk):
                      names.append(m.group(1))
              return sorted(set(names))[-20:]

          def route_strings(text):
              pats = [
                  r"""["']([^"'\s]{1,180}\.(?:aspx|asmx|ashx|js)(?:\?[^"'\s]*)?)["']""",
                  r"""["'](/[^"'\s]{1,180})["']"""
              ]
              vals=set()
              for pat in pats:
                  for m in re.finditer(pat,text,re.I):
                      v=m.group(1)
                      if len(v)<=180:
                          vals.add(v)
              return sorted(vals)[:500]

          def function_names(text):
              pats=[
                  r"function\s+([A-Za-z_$][A-Za-z0-9_$]*)\s*\(",
                  r"([A-Za-z_$][A-Za-z0-9_$.]*)\s*=\s*function\s*\(",
                  r"([A-Za-z_$][A-Za-z0-9_$.]*)\s*:\s*function\s*\("
              ]
              vals=set()
              for pat in pats:
                  vals.update(m.group(1) for m in re.finditer(pat,text))
              return sorted(vals)[:2000]

          out={
              "task":"TASK_219M_TDA_JS_STATIC_MINING",
              "issue":715,
              "base_main_sha":"715c2a5369452c1a79e8db4235d1b747e1638f00",

                  "known_paths":PATHS,

          texts={}
          for path in PATHS:
              url=f"https://{HOST}{path}"
              request_count += 1
              req=urllib.request.Request(
                  url,
                  method="GET",
                  headers={"User-Agent":"Mozilla/5.0 TASK219M-public-script-static-mining"}
              )
              item={"path":path,"request_ordinal":request_count}
              try:
                  with opener.open(req, timeout=30) as resp:
                      status=getattr(resp,"status",None) or resp.getcode()
                      body=resp.read()
                      final_url=resp.geturl()
                      ctype=resp.headers.get("Content-Type","")
              except urllib.error.HTTPError as e:
                  status=e.code
                  body=e.read() if hasattr(e,"read") else b""
                  final_url=e.geturl()
                  ctype=(e.headers or {}).get("Content-Type","") if getattr(e,"headers",None) else ""
              except Exception as exc:
                  item.update({"status":"TRANSPORT_ERROR","error_type":type(exc).__name__})
                  out["resources"].append(item)
                  continue

              parsed=urlparse(final_url)
              item.update({
                  "http_status":status,
                  "final_host":parsed.hostname,
                  "final_path":parsed.path,
                  "redirect_followed":final_url != url,
                  "content_type":ctype,
                  "bytes":len(body),
                  "sha256":hashlib.sha256(body).hexdigest()
              })

              if status != 200 or (parsed.hostname or "").lower()!=HOST or final_url != url:
                  item["status"]="STOP_NONCANONICAL_RESPONSE"
                  out["resources"].append(item)
                  continue

              raw_path=Path("runtime/raw")/safe_name(path)
              raw_path.write_bytes(body)
              text=body.decode("utf-8",errors="replace")
              texts[path]=text

              occ={}
              hits=[]
              for target in TARGETS:
                  count=0
                  for m in re.finditer(re.escape(target),text,re.I):
                      count += 1
                      if len(hits)<250:
                          hits.append({
                              "target":target,
                              "line":line_of(text,m.start()),
                              "context_sha256":hashlib.sha256(bounded_context(text,m.start(),m.end()).encode("utf-8")).hexdigest(),
                              "context":bounded_context(text,m.start(),m.end()),
                              "nearby_symbols":symbol_window(text,m.start())
                          })
                  occ[target]=count
              item.update({
                  "status":"PASS_PUBLIC_SCRIPT",
                  "target_occurrences":occ,
                  "function_names":function_names(text),
                  "route_strings":route_strings(text),
                  "bounded_hits":hits
              })
              out["resources"].append(item)

          for target in TARGETS:
              carriers=[]
              for item in out["resources"]:
                  n=(item.get("target_occurrences") or {}).get(target,0)
                  if n:
                      carriers.append({"path":item["path"],"count":n})
              out["cross_resource_target_index"][target]=carriers

          # Structural relationships based on co-occurrence within bounded context.
          rel_targets=["LayerInfo","awsgetcontentareas.aspx","filterarea","runafter","updatemain","changetitle","showfilter","refreshtime"]
          for item in out["resources"]:
              if item.get("status")!="PASS_PUBLIC_SCRIPT":
                  continue

              for m in re.finditer(r"awsgetcontentareas\.aspx",text,re.I):
                  ctx=bounded_context(text,m.start(),m.end(),radius=1800)
                  present=[t for t in rel_targets if re.search(re.escape(t),ctx,re.I)]
                  out["candidate_relationships"].append({
                      "path":item["path"],
                      "anchor":"awsgetcontentareas.aspx",
                      "line":line_of(text,m.start()),
                      "cooccurring_contract_terms":sorted(set(present)),
                      "context_sha256":hashlib.sha256(ctx.encode("utf-8")).hexdigest(),
                      "context":ctx[:1600],
                      "nearby_symbols":symbol_window(text,m.start())
                  })

          out["summary"]={
              "request_count":request_count,
              "pass_resource_count":sum(1 for x in out["resources"] if x.get("status")=="PASS_PUBLIC_SCRIPT"),
