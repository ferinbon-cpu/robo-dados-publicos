      - name: Python 3.12

          python - <<'PY'

      - name: One bounded Drive read and ephemeral digest
        id: live
        env:
          GOOGLE_DRIVE_CLIENT_ID: ${{ secrets.GOOGLE_DRIVE_CLIENT_ID }}
          GOOGLE_DRIVE_CLIENT_SECRET: ${{ secrets.GOOGLE_DRIVE_CLIENT_SECRET }}
          GOOGLE_DRIVE_REFRESH_TOKEN: ${{ secrets.GOOGLE_DRIVE_REFRESH_TOKEN }}
          TASK091_RUN_ID: ${{ github.run_id }}
          TASK091_RUN_ATTEMPT: ${{ github.run_attempt }}
          TASK091_HEAD_SHA: ${{ github.sha }}
          TASK091_RUNNER_TEMP: ${{ runner.temp }}
        run: |
          python - <<'PY'
          from __future__ import annotations
          import hashlib
          import json
          import os
          from pathlib import Path
          from urllib.parse import parse_qs, urlparse
          from urllib.request import urlopen
          import shutil

          from robo_dados_publicos.storage.drive_rest import (
              DriveRESTClient,
              OAuthCredentials,
              TokenProvider,
          )
          from robo_dados_publicos.manual_ingest.ephemeral_runtime_digest import (
              REMOTE_EFFECT_KEYS,
              run_ephemeral_digest,
          )
          from robo_dados_publicos.manual_ingest.source_family_maturity import (
              load_maturity_registry,
          )

          FILE_ID = "1JTpCPj4_rL08RubO5wOVvBHjuwqKWfQ8"
          EXPECTED_BYTES = 17615179
          EXPECTED_SHA256 = "44d92a6ac948bbf43dcb3302733faac1a4ed5e592702f66c07f0c6ede4ecb73c"
          EXPECTED_COUNTS = {"silver_rows": 79, "gold_rows": 15, "rag_rows": 126}
          root = Path(os.environ["TASK091_RUNNER_TEMP"]) / "task091"
          workspace = root / "workspace"
          evidence_dir = root / "evidence"
          workspace.mkdir(parents=True, exist_ok=False)
          evidence_dir.mkdir(parents=True, exist_ok=False)
          evidence_path = evidence_dir / "result.json"

          class GuardedOpener:
              def __init__(self):
                  self.records = []

              def __call__(self, req, timeout=0):
                  method = req.get_method()
                  parsed = urlparse(req.full_url)
                  qs = parse_qs(parsed.query)
                  ordinal = len(self.records) + 1
                  if ordinal > 2:
                      raise RuntimeError("STOP_TASK091_REQUEST_BUDGET")
                  if ordinal == 1:
                      if not (
                          method == "POST"
                          and parsed.scheme == "https"
                          and parsed.netloc == "oauth2.googleapis.com"
                          and parsed.path == "/token"
                      ):
                          raise RuntimeError("STOP_TASK091_OAUTH_ROUTE")
                      route_class = "OAUTH_TOKEN_EXCHANGE"
                  else:
                      expected_path = f"/drive/v3/files/{FILE_ID}"
                      if not (
                          method == "GET"
                          and parsed.scheme == "https"
                          and parsed.netloc == "www.googleapis.com"
                          and parsed.path == expected_path
                          and qs == {"alt": ["media"]}
                      ):
                          raise RuntimeError("STOP_TASK091_DRIVE_ROUTE")
                      route_class = "EXACT_DRIVE_MEDIA_GET"
                  self.records.append({
                      "ordinal": ordinal,
                      "method": method,
                      "host": parsed.netloc,
                      "route_class": route_class,
                  })
                  return urlopen(req, timeout=timeout)

          guard = GuardedOpener()
          evidence = {
              "task": "TASK_091_LIVE_EPHEMERAL_DRIVE_DIGEST",
              "run": {
                  "id": os.environ["TASK091_RUN_ID"],
                  "attempt": os.environ["TASK091_RUN_ATTEMPT"],
                  "head_sha": os.environ["TASK091_HEAD_SHA"],
              },
              "source": {
                  "drive_file_id": FILE_ID,
                  "expected_bytes": EXPECTED_BYTES,
                  "expected_sha256": EXPECTED_SHA256,
              },
              "authorization": {
                  "owner_instruction": "2026-09-04: prossiga",
                  "single_use": True,
                  "retry_authorized": False,
                  "future_execution_authorized": False,
              },
              "persistent_effects": {
                  key: 0 for key in REMOTE_EFFECT_KEYS
              },
          }

          exit_code = 0
          try:
              creds = OAuthCredentials.from_env()
              client = DriveRESTClient(TokenProvider(creds, opener=guard), opener=guard)
              source_path = workspace / "input.pdf"
              download = client.get(FILE_ID, source_path)

              if len(guard.records) != 2:
                  raise RuntimeError("STOP_TASK091_REQUEST_COUNT")
              if download["bytes"] != EXPECTED_BYTES:
                  raise RuntimeError("STOP_TASK091_SOURCE_BYTES")
              if download["sha256"] != EXPECTED_SHA256:
                  raise RuntimeError("STOP_TASK091_SOURCE_SHA256")

              manifest = {
                  "schema": "EPHEMERAL_DIGEST_BATCH_V1",
                  "batch_id": "task091-live-7024",
                  "inputs": [
                      {
                          "family": "JORNAL_OFICIAL",
                          "source_key": "journal-7024",
                          "relative_path": "input.pdf",
                          "mime_type": "application/pdf",
                          "metadata": {
                              "edition": 7024,
                              "publication_date": "2025-07-08",
                              "source_url": None,
                          },
                      }
                  ],
                  "remote_effects_authorized": {
                      key: False for key in REMOTE_EFFECT_KEYS
                  },
              }
              contract = json.loads(
                  Path("config/ephemeral_runtime_digest.v1.json").read_text(encoding="utf-8")
              )
              maturity = load_maturity_registry(
                  Path("config/source_family_maturity_registry.v1.json")
              )
              result = run_ephemeral_digest(
                  contract,
                  manifest,
                  maturity,
                  workspace_root=workspace,
              )

              if result["status"] != "PASS_EPHEMERAL_RUNTIME_DIGEST_NOT_PERSISTED":
                  raise RuntimeError("STOP_TASK091_DIGEST_STATUS")
              if result["input_count"] != 1 or result["candidate_file_count"] != 4:
                  raise RuntimeError("STOP_TASK091_DIGEST_FILE_COUNTS")
              item = result["items"][0]
              observed_counts = {
                  "silver_rows": item["silver_rows"],
                  "gold_rows": item["gold_rows"],
                  "rag_rows": item["rag_rows"],
              }
              if observed_counts != EXPECTED_COUNTS:
                  raise RuntimeError("STOP_TASK091_HISTORICAL_COUNT_DRIFT")
              if any(result["effects"][key] != 0 for key in REMOTE_EFFECT_KEYS):
                  raise RuntimeError("STOP_TASK091_REMOTE_EFFECT_COUNTER")

              evidence.update({
                  "status": "PASS_TASK091_LIVE_DRIVE_TO_EPHEMERAL_DIGEST",
                  "requests": guard.records,
                  "request_count": len(guard.records),
                  "observed_source": {
                      "bytes": download["bytes"],
                      "sha256": download["sha256"],
                  },
                  "digest": {
                      "status": result["status"],
                      "result_sha256": result["result_sha256"],
                      "candidate_set_sha256": result["candidate_set_sha256"],
                      "candidate_file_count": result["candidate_file_count"],
                      "processor_git_blob_sha": result["processor_git_blob_sha"],
                      "counts": observed_counts,
                      "historical_task073_counts_match": True,
                      "candidate_files": [
                          {
                              "name": f["name"],
                              "bytes": f["bytes"],
                              "sha256": f["sha256"],
                          }
                          for f in item["candidate_files"]
                      ],
                  },
              })
          except Exception as exc:
              exit_code = 2
              evidence.update({
                  "status": "STOP_TASK091_LIVE_DRIVE_TO_EPHEMERAL_DIGEST",
                  "reason": f"{type(exc).__name__}:{str(exc)}",
                  "requests": guard.records,
                  "request_count": len(guard.records),
              })
          finally:
              shutil.rmtree(workspace, ignore_errors=True)
              evidence["ephemeral_cleanup_verified"] = not workspace.exists()
              raw = json.dumps(evidence, ensure_ascii=False, indent=2, sort_keys=True) + "\n"
              evidence["evidence_payload_sha256_without_self_hash"] = hashlib.sha256(raw.encode("utf-8")).hexdigest()
              evidence_path.write_text(
                  json.dumps(evidence, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
                  encoding="utf-8",
              )

          print(json.dumps({
              "status": evidence["status"],
              "request_count": evidence["request_count"],
              "ephemeral_cleanup_verified": evidence["ephemeral_cleanup_verified"],
          }, sort_keys=True))
          raise SystemExit(exit_code)
          PY

      - name: Upload sanitized TASK 091 evidence only
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: task-091-live-ephemeral-digest-evidence
          path: ${{ runner.temp }}/task091/evidence/result.json
          if-no-files-found: error
          retention-days: 1
