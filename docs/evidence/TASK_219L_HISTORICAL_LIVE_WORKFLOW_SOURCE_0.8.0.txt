          def norm(s):
              s = unicodedata.normalize("NFKD", str(s or ""))
              s = "".join(ch for ch in s if not unicodedata.combining(ch))
              s = s.casefold()
              s = re.sub(r"[^a-z0-9]+", "_", s).strip("_")
              return s

          def safe_identifier(value):
              s = str(value or "")
              if not SAFE_ID.fullmatch(s):
                  return None
              if re.search(r"\d{7,}", s):
                  return None
              low = norm(s)
              if any(x in low for x in ("token","session","auth","password","secret","cookie","captcha","credential")):
                  return None
              return s

          def bucket(n):
              if n == 0: return "0"
              if n <= 16: return "1-16"
              if n <= 64: return "17-64"
              if n <= 256: return "65-256"
              if n <= 1024: return "257-1024"
              if n <= 4096: return "1025-4096"
              if n <= 16384: return "4097-16384"
              if n <= 65536: return "16385-65536"
              return "65537+"

          def schema_digest(obj):
              raw = json.dumps(obj, ensure_ascii=False, sort_keys=True, separators=(",",":")).encode("utf-8")
              return hashlib.sha256(raw).hexdigest()

          def scalar_type(v):
              if v is None: return "NULL"
              if isinstance(v, bool): return "BOOL"
              if isinstance(v, int) and not isinstance(v, bool): return "INT"
              if isinstance(v, float): return "DECIMAL"
              if isinstance(v, str): return "TEXT"
              return type(v).__name__.upper()

          def object_schema(obj, identifiers, depth=0):
              if depth >= MAX_DEPTH:
                  return {"type":"DEPTH_LIMIT"}
              if isinstance(obj, dict):
                  fields = {}
                  for k in sorted(obj, key=lambda x: str(x))[:300]:
                      sk = safe_identifier(k)
                      if sk:
                          identifiers.add(sk)
                          key = sk
                      else:
                          key = "UNSAFE_KEY_SHA256_" + hashlib.sha256(str(k).encode("utf-8", errors="replace")).hexdigest()[:16]
                      fields[key] = object_schema(obj[k], identifiers, depth+1)
                  return {
                      "type":"OBJECT",
                      "key_count":len(obj),
                      "keys_truncated":len(obj)>300,
                      "fields":fields,
                  }
              if isinstance(obj, (list, tuple)):
                  samples = [object_schema(v, identifiers, depth+1) for v in obj[:100]]
                  canon = sorted(set(json.dumps(x, ensure_ascii=False, sort_keys=True, separators=(",",":")) for x in samples))
                  return {
                      "type":"LIST",
                      "length":len(obj),
                      "sample_truncated":len(obj)>100,
                      "unique_element_schema_count":len(canon),
                      "element_schemas":[json.loads(x) for x in canon[:20]],
                  }
              out = {"type":scalar_type(obj)}
              if isinstance(obj, str):
                  out["length_bucket"] = bucket(len(obj))
              return out

          class StructuralHTMLParser(HTMLParser):
              def __init__(self):
                  super().__init__(convert_charrefs=True)
                  self.tags = Counter()
                  self.attrs = Counter()
                  self.controls = set()
                  self.ids = set()
                  self.classes = set()
              def handle_starttag(self, tag, attrs):
                  t = (tag or "").lower()
                  self.tags[t] += 1
                  amap = {}
                  for k,v in attrs:
                      kk = (k or "").lower()
                      self.attrs[kk] += 1
                      amap[kk] = v
                  if t in {"input","select","textarea","button","form","table","option"}:
                      nm = safe_identifier(amap.get("name"))
                      ty = safe_identifier(amap.get("type"))
                      if nm or ty:
                          self.controls.add((t,nm,ty))
                  sid = safe_identifier(amap.get("id"))
                  if sid:
                      self.ids.add(sid)
                  cls = str(amap.get("class") or "")
                  for token in re.split(r"\s+", cls):
                      st = safe_identifier(token)
                      if st:
                          self.classes.add(st)

          def html_schema(text):
              p = StructuralHTMLParser()
              try:
                  p.feed(text)
              except Exception:
                  pass
              if not p.tags:
                  return None
              identifiers = set()
              identifiers.update(x for x in p.ids if x)
              identifiers.update(x for x in p.classes if x)
              for _,nm,ty in p.controls:
                  if nm: identifiers.add(nm)
                  if ty: identifiers.add(ty)
              schema = {
                  "type":"HTML_FRAGMENT",
                  "tag_counts":dict(sorted(p.tags.items())),
                  "attribute_name_counts":dict(sorted(p.attrs.items())),
                  "controls":[
                      {"tag":t,"name":nm,"type":ty}

                  ][:150],
                  "safe_id_count":len(p.ids),
                  "safe_class_count":len(p.classes),
              }
              return schema, sorted(identifiers)[:MAX_IDS]

          def xml_schema(text):
              try:
                  root = ET.fromstring(text)
              except Exception:
                  return None
              tags=Counter(); attrs=Counter(); ids=set()
              def walk(el, depth=0):
                  if depth > MAX_DEPTH: return
                  st = safe_identifier(el.tag)
                  if st:
                      tags[st]+=1; ids.add(st)
                  for k in el.attrib:
                      sk=safe_identifier(k)
                      if sk:
                          attrs[sk]+=1; ids.add(sk)
                  for ch in list(el):
                      walk(ch, depth+1)
              walk(root)
              schema={"type":"XML","tag_counts":dict(sorted(tags.items())),"attribute_name_counts":dict(sorted(attrs.items()))}
              return schema, sorted(ids)[:MAX_IDS]

          def lexical_js_keys(text):
              found=set()
              for pat in (
                  r'["\']([A-Za-z_][A-Za-z0-9_.:\-]{0,79})["\']\s*:',
                  r'(?<![A-Za-z0-9_])([A-Za-z_][A-Za-z0-9_.:\-]{0,79})\s*:'
              ):
                  for m in re.finditer(pat, text[:500000]):
                      sk=safe_identifier(m.group(1))
                      if sk: found.add(sk)
                      if len(found)>=MAX_IDS: return sorted(found)
              return sorted(found)

          def classify_nested(value):
              text = str(value or "")
              base = {
                  "empty": text == "",
                  "length_bucket": bucket(len(text)),
                  "raw_value_persisted": False,
                  "visible_text_persisted": False,
              }
              if text == "":
                  schema={"type":"EMPTY"}
                  return {**base,"encoding_family":"EMPTY","schema":schema,"schema_sha256":schema_digest(schema),"structural_identifiers":[]}

              stripped=text.strip()

                  obj=json.loads(stripped)
                  ids=set()
                  schema=object_schema(obj, ids)
                  return {**base,"encoding_family":"JSON","schema":schema,"schema_sha256":schema_digest(schema),"structural_identifiers":sorted(ids)[:MAX_IDS]}
              except Exception:
                  pass

              try:
                  obj=ast.literal_eval(stripped)
                  if isinstance(obj,(dict,list,tuple)):
                      ids=set()
                      schema=object_schema(obj,ids)
                      return {**base,"encoding_family":"PYTHON_LITERAL_SAFE","schema":schema,"schema_sha256":schema_digest(schema),"structural_identifiers":sorted(ids)[:MAX_IDS]}
              except Exception:
                  pass

              xs=xml_schema(stripped)
              if xs:
                  schema,ids=xs

              hs=html_schema(stripped)

              keys=lexical_js_keys(stripped)
              if keys:
                  schema={"type":"OPAQUE_OR_JS_TEXT","lexical_key_names":keys,"lexical_key_count":len(keys)}
                  return {**base,"encoding_family":"OPAQUE_OR_JS_TEXT_WITH_KEYS","schema":schema,"schema_sha256":schema_digest(schema),"structural_identifiers":keys}

              schema={"type":"OPAQUE_TEXT","length_bucket":bucket(len(stripped))}
              return {**base,"encoding_family":"OPAQUE_TEXT","schema":schema,"schema_sha256":schema_digest(schema),"structural_identifiers":[]}
